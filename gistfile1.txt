    <script>
        // Taux de conversion statiques indicatifs (à ajuster selon le marché actuel)
        const Taux_Euro = 655.957; 
        const Taux_Dollar = 600.00; 

        // 1. Protection simple de l'espace Admin par mot de passe
        function accessAdminPanel() {
            let password = prompt("Veuillez saisir votre code d'accès propriétaire :");
            // Remplacez 'admin123' par le mot de passe de votre choix
            if (password === 'admin123') {
                alert("Accès autorisé. Le panneau de téléversement est maintenant visible.");
                document.getElementById('admin-panel').style.display = 'block';
                document.getElementById('admin-panel').scrollIntoView({ behavior: 'smooth' });
            } else {
                alert("Accès refusé. Réservé à l'administrateur.");
            }
        }

        // 2. Aperçu des conversions en temps réel dans le formulaire
        function previewConversions() {
            const fcfa = parseFloat(document.getElementById('book-price-input').value);
            const preview = document.getElementById('conversion-preview');
            if (fcfa > 0) {
                let euro = (fcfa / Taux_Euro).toFixed(2);
                let dollar = (fcfa / Taux_Dollar).toFixed(2);
                preview.textContent = `Aperçu conversion : ~ ${euro} € | ~ ${dollar} $`;
            } else {
                preview.textContent = "";
            }
        }

        // 3. Logique d'ajout dynamique d'un livre sur la page
        function addNewBook(event) {
            event.preventDefault();

            // Récupération des données du formulaire
            const title = document.getElementById('book-title-input').value;
            const cover = document.getElementById('book-cover-input').value;
            const fcfa = parseFloat(document.getElementById('book-price-input').value);
            const comment = document.getElementById('book-comment-input').value;

            // Calcul des conversions automatiques
            const euro = (fcfa / Taux_Euro).toFixed(2);
            const dollar = (fcfa / Taux_Dollar).toFixed(2);

            // Formatage du prix en FCFA avec espaces pour la lisibilité
            const formattedFCFA = new Intl.NumberFormat('fr-FR').format(fcfa) + " FCFA";

            // Encodage du texte pour le lien WhatsApp automatique
            const whatsappMessage = encodeURIComponent(`Bonjour L'Âme Active, je souhaite commander le livre : ${title}`);
            const whatsappLink = `https://wa.me/22896342801?text=${whatsappMessage}`;

            // Création de la carte HTML du livre
            const bookCard = document.createElement('div');
            bookCard.className = 'book-card';
            bookCard.innerHTML = `
                <div class="book-cover" style="background-image: url('${cover}');"></div>
                <div class="book-info">
                    <h3 class="book-title">${title}</h3>
                    <p class="book-commentary">"${comment}"</p>
                    <div class="books-prices">
                        <span class="price-fcf">${formattedFCFA}</span>
                        <div class="price-converted">
                            <span><i class="fa-solid fa-euro-sign"></i> ${euro} €</span>
                            <span><i class="fa-solid fa-dollar-sign"></i> ${dollar} $</span>
                        </div>
                    </div>
                    <a href="${whatsappLink}" class="btn-whatsapp" target="_blank">
                        <i class="fa-brands fa-whatsapp"></i> Commander via WhatsApp
                    </a>
                </div>
            `;

            // Insertion au début de la liste
            const container = document.getElementById('books-container');
            container.insertBefore(bookCard, container.firstChild);

            // Réinitialisation du formulaire et fermeture du panneau admin
            document.getElementById('upload-form').reset();
            document.getElementById('conversion-preview').textContent = "";
            document.getElementById('admin-panel').style.display = 'none';
            
            alert(`Le livre "${title}" a été ajouté avec succès avec vos commentaires et prix convertis !`);
        }
    </script>
